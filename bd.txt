-- 0. CONFIGURACIÓN INICIAL (Extensiones y Enums)
-- En Supabase moderno ya viene pre-habilitada gen_random_uuid(), no es necesaria la extensión uuid-ossp.
CREATE TYPE rol_usuario AS ENUM ('ADMIN', 'CASERA', 'CIUDADANO');

CREATE TYPE unidad_medida_andina AS ENUM (
    'libra', 'cuartilla', 'arroba', 'cuarta', 'amarro', 'kilo', 'unidad', 'docena', 'carga', 'litro'
);

CREATE TYPE estado_procesamiento AS ENUM ('PENDIENTE', 'PROCESANDO', 'COMPLETADO', 'ERROR');
CREATE TYPE estado_moderacion AS ENUM ('PENDIENTE', 'APROBADO', 'RECHAZADO', 'SUGERIDO_IA');

-- 1. PERFILES (Sincronizado automáticamente con Auth de Supabase)
CREATE TABLE perfiles (
    id UUID PRIMARY KEY, -- Id del usuario en auth.users
    rol rol_usuario NOT NULL DEFAULT 'CIUDADANO',
    nombre_completo VARCHAR(150),
    creado_en TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 2. MERCADOS (Infraestructura de La Paz)
CREATE TABLE mercados (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    nombre VARCHAR(100) NOT NULL,
    zona VARCHAR(100) NOT NULL,
    direccion TEXT,
    dias_atencion_normal VARCHAR(150) DEFAULT 'Lunes a Domingo',
    dias_feria_ampliada VARCHAR(150), 
    horario_apertura TIME DEFAULT '06:00:00',
    horario_cierre TIME DEFAULT '18:00:00',
    latitud DECIMAL(10, 8),
    longitud DECIMAL(11, 8),
    reputacion_infraestructura DECIMAL(3, 2) DEFAULT 5.00, -- Calculado por Trigger
    activo BOOLEAN DEFAULT true,
    creado_por_admin UUID REFERENCES perfiles(id),
    creado_en TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 3. CASERAS (Los Comerciantes - Alineado con Frontend)
CREATE TABLE caseras (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    perfil_id UUID REFERENCES perfiles(id) ON DELETE CASCADE,
    nombre VARCHAR(100) NOT NULL,
    mercado_id UUID REFERENCES mercados(id) NOT NULL,
    nro_puesto VARCHAR(20),
    especialidad VARCHAR(100), 
    horario VARCHAR(100) DEFAULT 'Mismo del mercado',
    reputacion_score DECIMAL(3, 2) DEFAULT 5.00, -- Calculado por Trigger
    votos_peso_justo INT DEFAULT 0, -- Calculado por Trigger
    votos_peso_dudoso INT DEFAULT 0, -- Calculado por Trigger
    madrina_id UUID REFERENCES caseras(id) ON DELETE SET NULL, -- Sistema Madrina (Inclusión)
    apadrinada BOOLEAN DEFAULT false,
    sector VARCHAR(100),
    dias_dictado_consecutivo INT DEFAULT 0, -- Incentivos tributarios/gremiales
    creado_en TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 4. CATÁLOGO MAESTRO DE PRODUCTOS
CREATE TABLE productos_maestro (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    nombre VARCHAR(100) NOT NULL UNIQUE,
    categoria VARCHAR(50),
    peso_estandar_kg DECIMAL(10, 4), -- Para estandarizar precio de búsqueda por "Más Barato"
    icono_generico TEXT NOT NULL
);

-- 5. COLA ASÍNCRONA DE IA (Ingesta por Voz)
CREATE TABLE cola_procesamiento_ia (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    casera_id UUID REFERENCES caseras(id) ON DELETE CASCADE,
    audio_url TEXT NOT NULL,
    estado estado_procesamiento DEFAULT 'PENDIENTE',
    intentos INT DEFAULT 0,
    creado_en TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    procesado_en TIMESTAMP WITH TIME ZONE
);

-- 6. INVENTARIO VIVO (Precios Actuales)
CREATE TABLE inventario (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    casera_id UUID REFERENCES caseras(id) ON DELETE CASCADE,
    producto_id UUID REFERENCES productos_maestro(id) ON DELETE CASCADE,
    precio DECIMAL(10, 2) NOT NULL,
    unidad_medida unidad_medida_andina NOT NULL,
    precio_estandar_kilo DECIMAL(10, 2),
    version_lock INT DEFAULT 1,
    disponible BOOLEAN DEFAULT true,
    ultima_actualizacion TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    UNIQUE(casera_id, producto_id)
);

-- 7. HISTORIAL DE PRECIOS (Dashboard Inflación GAMLP)
CREATE TABLE historial_precios (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    producto_id UUID REFERENCES productos_maestro(id) ON DELETE CASCADE,
    mercado_id UUID REFERENCES mercados(id) ON DELETE CASCADE,
    precio_registrado DECIMAL(10, 2) NOT NULL,
    unidad_medida unidad_medida_andina NOT NULL,
    precio_estandar_kilo DECIMAL(10, 2),
    creado_en TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 8. LISTAS DE COMPRAS Y RECORRIDO LIMPIO
CREATE TABLE listas_compras (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    ciudadano_id UUID REFERENCES perfiles(id) ON DELETE CASCADE,
    creado_en TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE TABLE detalle_lista (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    lista_id UUID REFERENCES listas_compras(id) ON DELETE CASCADE,
    inventario_id UUID REFERENCES inventario(id) ON DELETE CASCADE,
    comprado BOOLEAN DEFAULT false,
    creado_en TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 9. COMUNIDAD Y ALERTAS (Alineado con waze-alerts.ts)
CREATE TABLE alertas_waze (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    inventario_id UUID REFERENCES inventario(id) ON DELETE CASCADE,
    casera_id UUID REFERENCES caseras(id) ON DELETE CASCADE, -- Optimiza filtrado directo
    ciudadano_id UUID REFERENCES perfiles(id) ON DELETE SET NULL,
    tipo_alerta VARCHAR(50) CHECK (tipo_alerta IN ('agotado', 'precio_alto')),
    activa BOOLEAN DEFAULT true,
    creado_en TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 10A. EVALUACIÓN DE INFRAESTRUCTURA (Mercados)
CREATE TABLE evaluacion_mercados (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    mercado_id UUID REFERENCES mercados(id) ON DELETE CASCADE,
    ciudadano_id UUID REFERENCES perfiles(id) ON DELETE SET NULL,
    calificacion_estrellas INT CHECK (calificacion_estrellas >= 1 AND calificacion_estrellas <= 5) NOT NULL,
    etiquetas_rapidas TEXT[], 
    sugerencia TEXT, 
    estado_moderacion estado_moderacion DEFAULT 'APROBADO',
    creado_en TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 10B. EVALUACIÓN DE CASERAS (Comentarios / Reputación)
CREATE TABLE evaluacion_caseras (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    casera_id UUID REFERENCES caseras(id) ON DELETE CASCADE,
    ciudadano_id UUID REFERENCES perfiles(id) ON DELETE SET NULL,
    calificacion_estrellas INT CHECK (calificacion_estrellas >= 1 AND calificacion_estrellas <= 5) NOT NULL,
    peso_fiel BOOLEAN NOT NULL DEFAULT true, 
    etiquetas_rapidas TEXT[], 
    comentario TEXT, -- Si el comentario no es null, cuenta como reseña textual
    estado_moderacion estado_moderacion DEFAULT 'APROBADO',
    creado_en TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- ==========================================
-- ⚡ RENDIMIENTO (ÍNDICES)
-- ==========================================
CREATE INDEX idx_caseras_mercado ON caseras(mercado_id);
CREATE INDEX idx_inventario_casera ON inventario(casera_id);
CREATE INDEX idx_inventario_producto ON inventario(producto_id);
CREATE INDEX idx_evaluacion_casera ON evaluacion_caseras(casera_id);
CREATE INDEX idx_evaluacion_mercado ON evaluacion_mercados(mercado_id);
CREATE INDEX idx_alertas_casera ON alertas_waze(casera_id);
CREATE INDEX idx_historial_producto_mercado ON historial_precios(producto_id, mercado_id);

ALTER TABLE inventario ENABLE ROW LEVEL SECURITY;
ALTER TABLE evaluacion_mercados ENABLE ROW LEVEL SECURITY;
ALTER TABLE evaluacion_caseras ENABLE ROW LEVEL SECURITY;

-- Inventario Público (Cualquiera puede leer)
CREATE POLICY "Inventario publico" ON inventario FOR SELECT USING (true);

-- Las caseras insertan, actualizan y borran su propio inventario (o de su ahijada si es madrina)
CREATE POLICY "Caseras insertan su inventario o ahijadas" ON inventario FOR INSERT WITH CHECK (
    auth.uid() IN (
        SELECT perfil_id FROM caseras WHERE id = casera_id 
        UNION 
        SELECT madrina.perfil_id FROM caseras ahijada JOIN caseras madrina ON ahijada.madrina_id = madrina.id WHERE ahijada.id = casera_id
    )
);
CREATE POLICY "Caseras modifican su inventario o ahijadas" ON inventario FOR UPDATE USING (
    auth.uid() IN (
        SELECT perfil_id FROM caseras WHERE id = casera_id 
        UNION 
        SELECT madrina.perfil_id FROM caseras ahijada JOIN caseras madrina ON ahijada.madrina_id = madrina.id WHERE ahijada.id = casera_id
    )
);
CREATE POLICY "Caseras borran su inventario o ahijadas" ON inventario FOR DELETE USING (
    auth.uid() IN (
        SELECT perfil_id FROM caseras WHERE id = casera_id 
        UNION 
        SELECT madrina.perfil_id FROM caseras ahijada JOIN caseras madrina ON ahijada.madrina_id = madrina.id WHERE ahijada.id = casera_id
    )
);

-- Evaluaciones Públicas (Para que se vean en tu demo)
CREATE POLICY "Lectura publica mercados" ON evaluacion_mercados FOR SELECT USING (true);
CREATE POLICY "Lectura publica caseras" ON evaluacion_caseras FOR SELECT USING (true);

-- Solo usuarios logueados evalúan mercados
CREATE POLICY "Ciudadanos evaluan mercados" ON evaluacion_mercados FOR INSERT WITH CHECK (
    auth.uid() = ciudadano_id
);

-- 🔥 ANTI-TRAMPA: Usuarios logueados evalúan caseras, EXCEPTO la propia casera a sí misma
CREATE POLICY "Ciudadanos evaluan caseras" ON evaluacion_caseras FOR INSERT WITH CHECK (
    auth.uid() = ciudadano_id 
    AND auth.uid() NOT IN (
        SELECT perfil_id FROM caseras WHERE id = casera_id
    )
);

-- Hacer tablas principales públicas para lectura (en caso de que RLS esté activado globalmente)
ALTER TABLE mercados ENABLE ROW LEVEL SECURITY;
ALTER TABLE caseras ENABLE ROW LEVEL SECURITY;
ALTER TABLE productos_maestro ENABLE ROW LEVEL SECURITY;

-- ⚙️ AUTOMATIZACIONES Y TRIGGERS
-- ==========================================

-- A. Actualiza métricas de la CASERA automáticamente (Estrellas y Peso Fiel)
CREATE OR REPLACE FUNCTION actualizar_metricas_casera()
RETURNS TRIGGER AS $$
BEGIN
    UPDATE caseras 
    SET 
        reputacion_score = (SELECT COALESCE(ROUND(AVG(calificacion_estrellas), 2), 5.00) FROM evaluacion_caseras WHERE casera_id = NEW.casera_id),
        votos_peso_justo = (SELECT COUNT(*) FROM evaluacion_caseras WHERE casera_id = NEW.casera_id AND peso_fiel = true),
        votos_peso_dudoso = (SELECT COUNT(*) FROM evaluacion_caseras WHERE casera_id = NEW.casera_id AND peso_fiel = false)
    WHERE id = NEW.casera_id;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trigger_metricas_casera ON evaluacion_caseras;
CREATE TRIGGER trigger_metricas_casera
AFTER INSERT OR UPDATE ON evaluacion_caseras
FOR EACH ROW EXECUTE FUNCTION actualizar_metricas_casera();

-- B. Actualiza reputación del MERCADO automáticamente
CREATE OR REPLACE FUNCTION actualizar_reputacion_mercado()
RETURNS TRIGGER AS $$
BEGIN
    UPDATE mercados 
    SET reputacion_infraestructura = (SELECT COALESCE(ROUND(AVG(calificacion_estrellas), 2), 5.00) FROM evaluacion_mercados WHERE mercado_id = NEW.mercado_id)
    WHERE id = NEW.mercado_id;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trigger_reputacion_mercado ON evaluacion_mercados;
CREATE TRIGGER trigger_reputacion_mercado
AFTER INSERT OR UPDATE ON evaluacion_mercados
FOR EACH ROW EXECUTE FUNCTION actualizar_reputacion_mercado();

-- C. Registra el Historial de Precios para el GAMLP
CREATE OR REPLACE FUNCTION registrar_historial_precios()
RETURNS TRIGGER AS $$
DECLARE v_mercado_id UUID;
BEGIN
    SELECT mercado_id INTO v_mercado_id FROM caseras WHERE id = NEW.casera_id;
    INSERT INTO historial_precios (producto_id, mercado_id, precio_registrado, unidad_medida, precio_estandar_kilo)
    VALUES (NEW.producto_id, v_mercado_id, NEW.precio, NEW.unidad_medida, NEW.precio_estandar_kilo);
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trigger_registrar_historial ON inventario;
CREATE TRIGGER trigger_registrar_historial
AFTER INSERT OR UPDATE ON inventario
FOR EACH ROW EXECUTE FUNCTION registrar_historial_precios();

-- D. Creación automática de perfiles públicos al registrarse en Auth
CREATE OR REPLACE FUNCTION crear_perfil_por_defecto()
RETURNS TRIGGER AS $$
BEGIN
    INSERT INTO public.perfiles (id, rol, nombre_completo)
    VALUES (
        NEW.id, 
        COALESCE(UPPER(NEW.raw_user_meta_data->>'rol')::rol_usuario, 'CIUDADANO'), 
        COALESCE(NEW.raw_user_meta_data->>'nombre_completo', 'Usuario Nuevo')
    );
    RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

DROP TRIGGER IF EXISTS trigger_crear_perfil_nuevo ON auth.users;
CREATE TRIGGER trigger_crear_perfil_nuevo
AFTER INSERT ON auth.users
FOR EACH ROW EXECUTE FUNCTION crear_perfil_por_defecto();

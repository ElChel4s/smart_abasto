-- STREAMING_CHUNK:Configurando extensiones y tipos de datos (ENUMs)...
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- Tipos de roles para diferenciar a los usuarios en la app
CREATE TYPE rol_usuario AS ENUM ('ADMIN', 'CASERA', 'CIUDADANO');

-- Unidades paceñas para que la IA estandarice los dictados
CREATE TYPE unidad_medida_andina AS ENUM (
    'libra', 'cuartilla', 'arroba', 'cuarta', 'amarro', 'kilo', 'unidad', 'docena', 'carga'
);

-- Estados del filtro anti-agresividad (Gobernanza)
CREATE TYPE estado_moderacion AS ENUM ('PENDIENTE', 'APROBADO', 'RECHAZADO', 'SUGERIDO_IA');

-- NUEVO: Estado para la cola asíncrona de audios
CREATE TYPE estado_procesamiento AS ENUM ('PENDIENTE', 'PROCESANDO', 'COMPLETADO', 'ERROR');

-- STREAMING_CHUNK:Creando tabla de perfiles globales...
-- 👤 1. Tabla de Perfiles (Sincronizada automáticamente con auth.users mediante Trigger)
CREATE TABLE perfiles (
    id UUID PRIMARY KEY, -- El ID es provisto directamente por el UUID de auth.users
    rol rol_usuario NOT NULL DEFAULT 'CIUDADANO',
    nombre_completo VARCHAR(150),
    creado_en TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- STREAMING_CHUNK:Diseñando la estructura de Mercados y Horarios...
-- 📍 2. Tabla: Mercados de La Paz (Administrables)
CREATE TABLE mercados (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    nombre VARCHAR(100) NOT NULL,
    zona VARCHAR(100) NOT NULL,
    direccion TEXT,
    -- Horarios y días de feria (Ej: "Lunes a Viernes", "Feria ampliada Sábados y Domingos")
    dias_atencion_normal VARCHAR(150) DEFAULT 'Lunes a Domingo',
    dias_feria_ampliada VARCHAR(150), 
    horario_apertura TIME DEFAULT '06:00:00',
    horario_cierre TIME DEFAULT '18:00:00',
    latitud DECIMAL(10, 8),
    longitud DECIMAL(11, 8),
    activo BOOLEAN DEFAULT true,
    creado_por_admin UUID REFERENCES perfiles(id), -- El admin que lo registró
    creado_en TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- STREAMING_CHUNK:Definiendo a las Comerciantes y el modelo Madrina...
-- 👩‍🌾 3. Tabla: Comerciantes (Caseras)
CREATE TABLE comerciantes (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    perfil_id UUID REFERENCES perfiles(id) ON DELETE CASCADE, -- Vinculado al login de la casera
    nombre_casera VARCHAR(100) NOT NULL, -- Ej: "Doña Rosita"
    mercado_id UUID REFERENCES mercados(id) NOT NULL,
    nro_puesto VARCHAR(20),
    categoria VARCHAR(50), 
    -- Horarios personalizados de la casera (puede diferir del mercado)
    horario_personalizado VARCHAR(100) DEFAULT 'Mismo del mercado',
    reputacion_score DECIMAL(3, 2) DEFAULT 5.00,
    -- 🔥 MAGIA INCLUSIVA: El modelo "Casera Madrina"
    -- Si madrina_id tiene un valor, esta casera (abuelita) es administrada por otra (madrina)
    madrina_id UUID REFERENCES comerciantes(id), 
    dias_dictado_consecutivo INT DEFAULT 0, -- Incentivo tributario municipal
    creado_en TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- STREAMING_CHUNK:Armando el Catálogo Maestro con Íconos Genéricos...
-- 🥔 4. Tabla: Catálogo Maestro de Productos (Iconos genéricos, cero fotos)
CREATE TABLE productos_maestro (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    nombre VARCHAR(100) NOT NULL UNIQUE,
    categoria VARCHAR(50),
    -- NUEVO: Factor de conversión (Ej: Arroba = 11.5 kg) para estandarización interna
    peso_estandar_kg DECIMAL(10, 4), 
    icono_generico TEXT NOT NULL -- Ej: '🥔', '🍅', o nombre de un ícono SVG 'icon-papa'
);

-- NUEVO: Cola de procesamiento asíncrono para audios (Evita colapso a las 6:00 AM)
CREATE TABLE cola_procesamiento_ia (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    comerciante_id UUID REFERENCES comerciantes(id),
    audio_url TEXT NOT NULL,
    estado estado_procesamiento DEFAULT 'PENDIENTE',
    intentos INT DEFAULT 0,
    creado_en TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    procesado_en TIMESTAMP WITH TIME ZONE
);

-- STREAMING_CHUNK:Estructurando el Inventario Vivo...
-- 🥕 5. Tabla: Inventario Vivo (Precios actualizados por IA)
CREATE TABLE inventario (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    comerciante_id UUID REFERENCES comerciantes(id) ON DELETE CASCADE,
    producto_id UUID REFERENCES productos_maestro(id) ON DELETE CASCADE,
    precio DECIMAL(10, 2) NOT NULL,
    unidad_medida unidad_medida_andina NOT NULL,
    -- NUEVO: Normalización Matemática y Concurrencia (Defensas Técnicas)
    precio_estandar_kilo DECIMAL(10, 2), -- Calculado en backend. Permite ordenar por "Más Barato" sin importar si dictaron libra o cuarta.
    version_lock INT DEFAULT 1, -- Control de concurrencia optimista (Previene colisión de updates simultáneos)
    disponible BOOLEAN DEFAULT true,
    ultima_actualizacion TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    UNIQUE(comerciante_id, producto_id)
);

-- 📈 5D. Tabla: Historial de Precios para el Dashboard de Inflación del GAMLP
CREATE TABLE historial_precios (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    producto_id UUID REFERENCES productos_maestro(id) ON DELETE CASCADE,
    mercado_id UUID REFERENCES mercados(id) ON DELETE CASCADE,
    precio_registrado DECIMAL(10, 2) NOT NULL,
    unidad_medida unidad_medida_andina NOT NULL,
    precio_estandar_kilo DECIMAL(10, 2),
    creado_en TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- STREAMING_CHUNK:Creando tablas para la Lista de Compras del Ciudadano...
-- 👤 5B. Tabla: Cabecera de la Lista de Compras del Ciudadano
CREATE TABLE listas_compras (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    ciudadano_id UUID REFERENCES perfiles(id) ON DELETE CASCADE,
    creado_en TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 🛒 5C. Tabla: Detalle de la Lista (El enrutador físico lee esto)
CREATE TABLE detalle_lista (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    lista_id UUID REFERENCES listas_compras(id) ON DELETE CASCADE,
    inventario_id UUID REFERENCES inventario(id) ON DELETE CASCADE,
    comprado BOOLEAN DEFAULT false,
    creado_en TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- STREAMING_CHUNK:Separando las Alertas del Waze y los Comentarios...
-- 🚨 6A. Tabla: Alertas de Comunidad (El "Waze" de los mercados)
-- Sirve para notificar cosas rápidas, no afecta la reputación (estrellas) de la casera directamente.
CREATE TABLE alertas_waze (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    inventario_id UUID REFERENCES inventario(id) ON DELETE CASCADE,
    ciudadano_id UUID REFERENCES perfiles(id),
    tipo_alerta VARCHAR(50) CHECK (tipo_alerta IN ('AGOTADO', 'PRECIO_INCORRECTO')),
    activa BOOLEAN DEFAULT true,
    creado_en TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 🗣️ 6B. Tabla: Reseñas y Comentarios (El sistema de reputación con IA)
-- Pasa por el filtro anti-toxicidad antes de ser publicado.
CREATE TABLE resenas_caseras (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    comerciante_id UUID REFERENCES comerciantes(id) ON DELETE CASCADE,
    ciudadano_id UUID REFERENCES perfiles(id),
    calificacion INT CHECK (calificacion >= 1 AND calificacion <= 5),
    comentario TEXT,
    -- La IA de Gemini decidirá si el estado es APROBADO o SUGERIDO_IA (si era agresivo)
    estado_moderacion estado_moderacion DEFAULT 'PENDIENTE',
    comentario_sugerido_ia TEXT, -- Si fue agresivo, la IA guarda aquí la versión "educada"
    creado_en TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- STREAMING_CHUNK:Configurando Seguridad (RLS) y Tiempo Real...
-- 7. SEGURIDAD RLS (Row Level Security)
ALTER TABLE inventario ENABLE ROW LEVEL SECURITY;

-- Lectura pública para que cualquier ciudadano vea los precios
CREATE POLICY "Inventario publico" ON inventario FOR SELECT USING (true);

-- Escritura: Una casera solo actualiza su puesto o el de su "Ahijada" (Abuelita)
-- Eliminamos la política FOR ALL que causaba problemas con el INSERT
DROP POLICY IF EXISTS "Caseras modifican su inventario o ahijadas" ON inventario;

-- Política de Inserción (Permite a la casera añadir nuevas filas)
CREATE POLICY "Caseras insertan su inventario o ahijadas" ON inventario FOR INSERT WITH CHECK (
    auth.uid() IN (
        SELECT perfil_id FROM comerciantes WHERE id = comerciante_id
        UNION
        SELECT madrina.perfil_id FROM comerciantes ahijada
        JOIN comerciantes madrina ON ahijada.madrina_id = madrina.id
        WHERE ahijada.id = comerciante_id
    )
);

-- Política de Actualización (Permite a la casera modificar filas)
CREATE POLICY "Caseras modifican su inventario o ahijadas" ON inventario FOR UPDATE USING (
    auth.uid() IN (
        SELECT perfil_id FROM comerciantes WHERE id = comerciante_id
        UNION
        SELECT madrina.perfil_id FROM comerciantes ahijada
        JOIN comerciantes madrina ON ahijada.madrina_id = madrina.id
        WHERE ahijada.id = comerciante_id
    )
);

-- Política de Borrado (Permite a la casera eliminar filas)
CREATE POLICY "Caseras borran su inventario o ahijadas" ON inventario FOR DELETE USING (
    auth.uid() IN (
        SELECT perfil_id FROM comerciantes WHERE id = comerciante_id
        UNION
        SELECT madrina.perfil_id FROM comerciantes ahijada
        JOIN comerciantes madrina ON ahijada.madrina_id = madrina.id
        WHERE ahijada.id = comerciante_id
    )
);

-- STREAMING_CHUNK:Configurando Automatización (Triggers)...
-- 10. AUTOMATIZACIÓN DE REPUTACIÓN (TRIGGER)
-- Actualiza el score de la casera automáticamente cuando se aprueba una reseña
CREATE OR REPLACE FUNCTION calcular_reputacion_automatica()
RETURNS TRIGGER AS $$
BEGIN
    -- Evaluamos si se aprobó (ya sea en insert directo o al actualizar el estado)
    IF NEW.estado_moderacion = 'APROBADO' THEN
        UPDATE comerciantes 
        SET reputacion_score = (
            SELECT COALESCE(ROUND(AVG(calificacion), 2), 5.00) 
            FROM resenas_caseras 
            WHERE comerciante_id = NEW.comerciante_id AND estado_moderacion = 'APROBADO'
        )
        WHERE id = NEW.comerciante_id;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Mantenlo limpio vigilando el INSERT y UPDATE global, la función se encarga del filtro
DROP TRIGGER IF EXISTS trigger_actualizar_reputacion ON resenas_caseras;
CREATE TRIGGER trigger_actualizar_reputacion
AFTER INSERT OR UPDATE ON resenas_caseras
FOR EACH ROW EXECUTE FUNCTION calcular_reputacion_automatica();

-- 11. AUTOMATIZACIÓN DE HISTORIAL DE PRECIOS (TRIGGER INFLACIÓN GAMLP)
-- Guarda un registro histórico cada vez que un precio cambia
CREATE OR REPLACE FUNCTION registrar_historial_precios()
RETURNS TRIGGER AS $$
DECLARE
    v_mercado_id UUID;
BEGIN
    -- Obtenemos el mercado_id de la casera para indexar la inflación por zona
    SELECT mercado_id INTO v_mercado_id FROM comerciantes WHERE id = NEW.comerciante_id;

    INSERT INTO historial_precios (producto_id, mercado_id, precio_registrado, unidad_medida, precio_estandar_kilo)
    VALUES (NEW.producto_id, v_mercado_id, NEW.precio, NEW.unidad_medida, NEW.precio_estandar_kilo);
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trigger_registrar_historial ON inventario;
CREATE TRIGGER trigger_registrar_historial
AFTER INSERT OR UPDATE ON inventario
FOR EACH ROW EXECUTE FUNCTION registrar_historial_precios();
